abstract class QuranLocalDataSource {
  Future<List<AyahCoordinatesModel>> getCoordinates(int pageNumber, String mushafType);

class QuranLocalDataSourceImpl implements QuranLocalDataSource {
  @override
  Future<List<AyahCoordinatesModel>> getCoordinates(int pageNumber, String mushafType) async {
    try {
      // استخدام AssetsManager لجلب المسار الصحيح
      final path = AssetsManager.getCoordsJsonPath(
        mushafType: mushafType,
        pageNumber: pageNumber,
      );

      // قراءة الملف
      final String response = await rootBundle.loadString(path);
      final List<dynamic> data = json.decode(response);

      return data.map((e) => AyahCoordinatesModel.fromJson(e)).toList();
    } catch (e) {
      // في حال لم نجد الملف (مثل صفحات الفواصل أو خطأ في الاسم) نعيد قائمة فارغة
      print("QuranLocalDataSource Error: $e");
      return [];
    }
  }
}

class ReciterModel {
  final int id;
  final String nameArabic;
  final String slug;
  final String sourceUrl;
  // 1️⃣ الحقول الجديدة المضافة لتعريف جودة وتكوين الملفات
  final String bitRate;
  final String fileStructure;

  ReciterModel({
    required this.id,
    required this.nameArabic,
    required this.slug,
    required this.sourceUrl,
    required this.bitRate,
    required this.fileStructure,
  });

  // 2️⃣ Getter لسهولة العرض في الـ UI (مثل: 64kbps • Ayah)
  String get style => '$bitRate • $fileStructure';

  // 3️⃣ تحديث fromJson للتعامل مع البيانات الجديدة ومعالجة الرابط
  factory ReciterModel.fromJson(Map<String, dynamic> json) {
    final rawSource = json['source_url'] as String? ?? '';
    
    // التأكد من إزالة / في نهاية الرابط لتوحيد المعالجة لاحقاً
    final normalizedSource = rawSource.endsWith('/')
        ? rawSource.substring(0, rawSource.length - 1)
        : rawSource;

    return ReciterModel(
      id: json['id'] as int,
      nameArabic: json['name_arabic'] as String,
      slug: json['slug'] as String,
      sourceUrl: normalizedSource,
      // التعامل الآمن مع الحقول الجديدة في حال كانت null
      bitRate: (json['bit_rate'] as String?) ?? '',
      fileStructure: (json['file_structure'] as String?) ?? '',
    );
  }
}

import 'package:flutter_bloc/flutter_bloc.dart';
import '../../../../../core/services/audio_service.dart';
import '../../../domain/repositories/quran_repository.dart';
import 'quran_audio_event.dart';
import 'quran_audio_state.dart';

class QuranAudioBloc extends Bloc<QuranAudioEvent, QuranAudioState> {
  final AudioService _audioService;
  final QuranRepository _quranRepository;

  QuranAudioBloc(this._audioService, this._quranRepository) : super(const QuranAudioState()) {
    
    // 1. معالج تحميل القراء من الـ Repository
    on<LoadRecitersEvent>(_onLoadReciters);

    // 2. معالج تغيير القارئ (من القائمة أو الشيت)
    on<ChangeReciterEvent>(_onChangeReciter);

    // 3. معالج تشغيل المدى (للحفظ والتكرار)
    on<PlayAudioRangeEvent>(_onPlayAudioRange);

    // 4. معالجات التحكم في الصوت (بسيطة ومباشرة)
    on<PauseAudioEvent>((event, emit) => _audioService.pause());
    on<ResumeAudioEvent>((event, emit) => _audioService.resume());
    on<StopAudioEvent>((event, emit) => _audioService.stop());

    // الاستماع لتغيرات مشغل الصوت (لتحديث الـ UI لاحقاً إذا لزم الأمر)
    _audioService.playerStateStream.listen((playerState) {
       // يمكن إضافة منطق هنا لتغيير الحالة بناءً على playerState
       // مثلاً: عند انتهاء التلاوة نغير الحالة إلى stopped
    });
  }

  // ✅ منطق تحميل القراء النظيف
  Future<void> _onLoadReciters(LoadRecitersEvent event, Emitter<QuranAudioState> emit) async {
    try {
      // جلب القراء من الـ Repository مباشرة (بدون rootBundle هنا)
      final reciters = await _quranRepository.getReciters();
      
      emit(state.copyWith(
        reciters: reciters,
        // اختيار أول قارئ تلقائياً إذا لم يكن هناك قارئ مختار
        selectedReciter: state.selectedReciter ?? 
            (reciters.isNotEmpty ? reciters.first : null),
      ));
    } catch (e) {
      print("Error loading reciters: $e");
      // يفضل تخزين الخطأ في الحالة لعرضه للمستخدم
      emit(state.copyWith(errorMessage: "فشل تحميل قائمة القراء: $e"));
    }
  }

  // ✅ منطق تغيير القارئ وتحديث الحالة
  void _onChangeReciter(ChangeReciterEvent event, Emitter<QuranAudioState> emit) {
    emit(state.copyWith(selectedReciter: event.reciter));
  }

  // ✅ منطق تشغيل المدى (الآيات) المحدّث
  Future<void> _onPlayAudioRange(PlayAudioRangeEvent event, Emitter<QuranAudioState> emit) async {
    // 1. التحقق من وجود قارئ مختار
    if (state.selectedReciter == null) {
      emit(state.copyWith(
        status: AudioStatus.error, 
        errorMessage: "يرجى اختيار قارئ أولاً"
      ));
      return;
    }
    
    emit(state.copyWith(status: AudioStatus.loading, errorMessage: null));

    try {
      // 2. إيقاف أي تشغيل سابق لضمان عدم تداخل الأصوات
      await _audioService.stop();

      // 3. البدء في التشغيل باستخدام AudioService
      await _audioService.playRange(
        surahNumber: event.surahNumber,
        startAyah: event.startAyah,
        endAyah: event.endAyah,
        baseUrl: state.selectedReciter!.sourceUrl, // استخدام رابط القارئ المختار
        repeatAyah: event.repeatCount, // تكرار الفقرة كما جاء من MemorizationSheet
      );

      emit(state.copyWith(status: AudioStatus.playing));
    } catch (e) {
      emit(state.copyWith(status: AudioStatus.error, errorMessage: e.toString()));
    }
  }
}

import 'package:flutter_bloc/flutter_bloc.dart';
import '../../../domain/usecases/get_page_coordinates_usecase.dart';
import 'quran_overlay_event.dart';
import 'quran_overlay_state.dart';

class QuranOverlayBloc extends Bloc<QuranOverlayEvent, QuranOverlayState> {
  final GetPageCoordinatesUseCase _getPageCoordinatesUseCase;

  QuranOverlayBloc(this._getPageCoordinatesUseCase) : super(const QuranOverlayState()) {
    
    on<HighlightAyahEvent>((event, emit) async {
      // 1. جلب إحداثيات الصفحة كاملة
      final allCoordinates = await _getPageCoordinatesUseCase(
        pageNumber: event.pageNumber,
        mushafType: event.mushafType,
      );

      // 2. البحث عن الآية المطلوبة فقط
      // ملاحظة: الآية قد يكون لها أكثر من إحداثي (إذا كانت مقسومة على سطرين)
      final targetHighlights = allCoordinates
          .where((element) => element.ayahNumber == event.ayahNumber)
          .toList();

      // 3. إصدار حالة جديدة للرسم
      emit(QuranOverlayState(highlights: targetHighlights));
    });

    on<ClearHighlightEvent>((event, emit) {
      emit(const QuranOverlayState(highlights: []));
    });
  }
}

import 'package:flutter/material.dart';
import '../../domain/entities/ayah_coordinates.dart';

class HighlightOverlay extends StatelessWidget {
  final List<AyahCoordinates> highlights;
  final Size referenceSize; // حجم الصفحة الأصلي في ملف الـ JSON (للتحجيم)

  const HighlightOverlay({
    super.key,
    required this.highlights,
    // بناءً على ملف 001.json، العرض التقريبي للإحداثيات هو 230 والارتفاع 350
    // قد تحتاج لضبط هذه القيم بدقة بعد تجربة أول صفحة
    this.referenceSize = const Size(230, 350), 
  });

  @override
  Widget build(BuildContext context) {
    if (highlights.isEmpty) return const SizedBox.shrink();

    return CustomPaint(
      painter: _HighlightPainter(
        highlights: highlights,
        referenceSize: referenceSize,
      ),
      child: Container(),
    );
  }
}

class _HighlightPainter extends CustomPainter {
  final List<AyahCoordinates> highlights;
  final Size referenceSize;

  _HighlightPainter({required this.highlights, required this.referenceSize});

  @override
  void paint(Canvas canvas, Size size) {
    final Paint paint = Paint()
      ..color = Colors.yellow.withOpacity(0.4) // لون التظليل الشفاف
      ..style = PaintingStyle.fill;

    // حساب نسبة التكبير/التصغير لتناسب شاشة الجوال
    final double scaleX = size.width / referenceSize.width;
    final double scaleY = size.height / referenceSize.height;

    for (var element in highlights) {
      final path = _parsePolygon(element.polygonData, scaleX, scaleY);
      canvas.drawPath(path, paint);
    }
  }

  // تحويل النص "x,y x,y" إلى مسار رسم (Path)
  Path _parsePolygon(String polygonString, double scaleX, double scaleY) {
    final Path path = Path();
    final points = polygonString.split(' ');

    if (points.isNotEmpty) {
      final firstPoint = points[0].split(',');
      final startX = double.parse(firstPoint[0]) * scaleX;
      final startY = double.parse(firstPoint[1]) * scaleY;
      path.moveTo(startX, startY);

      for (int i = 1; i < points.length; i++) {
        final p = points[i].split(',');
        if (p.length == 2) {
          final x = double.parse(p[0]) * scaleX;
          final y = double.parse(p[1]) * scaleY;
          path.lineTo(x, y);
        }
      }
      path.close();
    }
    return path;
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => true;
}

  static const String _juzAssetPath =

    final List<dynamic> data = json.decode(jsonStr) as List<dynamic>;

      final j = e as Map<String, dynamic>;
      return _JuzInfo(
        id: j['id'] as int,
        firstPage: j['page_number'] as int,
      );
    }).toList();

    // ترتيب الأجزاء تصاعدياً
    list.sort((a, b) => a.id.compareTo(b.id));
    return list;
  }

/// شيت الحفظ والتكرار
/// - اختيار القارئ
/// - تحديد البدء (سورة + آية)

/// - تكرار الآية / تكرار الفقرة
/// - زر بدء
class MemorizationSheet extends StatefulWidget {
  final int currentSurah;
  final int currentPage;

  const MemorizationSheet({
    super.key,
    required this.currentSurah,
    required this.currentPage,

  State<MemorizationSheet> createState() => _MemorizationSheetState();

class _MemorizationSheetState extends State<MemorizationSheet> {
  // بيانات السور من surahs_index.json
  final List<_SurahInfo> _surahs = [];
  bool _isLoadingSurahs = true;

  int? _startSurahId;
  int? _endSurahId;
  int? _startAyah;
  int? _endAyah;

  int _ayahRepeatCount = 0; // تكرار الآية (UI فقط حالياً)
  int _rangeRepeatCount = 0; // تكرار الفقرة (المستخدم في PlayAudioRangeEvent)

  // خيارات التكرار
  final List<int> _repeatOptions = [0, 1, 3, 5, 10];

  // مسار ملف السور – عدِّله لو مختلف عندك
  static const String _surahsAssetPath =
      'assets/json/quran_data/surahs_index.json';

  @override
  void initState() {
    super.initState();
    _loadSurahs();
  }

  Future<void> _loadSurahs() async {
    try {
      final jsonStr = await rootBundle.loadString(_surahsAssetPath);
      final Map<String, dynamic> data =
          json.decode(jsonStr) as Map<String, dynamic>;
      final List<dynamic> chapters = data['chapters'] as List<dynamic>;

      _surahs.clear();
      for (final dynamic e in chapters) {
        final ch = e as Map<String, dynamic>;
        _surahInfoFromJson(ch);
      }

      _surahs.sort((a, b) => a.id.compareTo(b.id));

      // إعداد القيم الابتدائية
      final int defaultSurahId = widget.currentSurah;
      final _SurahInfo? defaultSurah =
          _surahs.firstWhere((s) => s.id == defaultSurahId, orElse: () {
        return _surahs.isNotEmpty ? _surahs.first : _SurahInfo.empty();
      });

      setState(() {
        _isLoadingSurahs = false;

        _startSurahId ??= defaultSurah.id;
        _endSurahId ??= defaultSurah.id;

        _startAyah ??= 1;
        _endAyah ??= defaultSurah.versesCount;
      });
    } catch (e) {
      setState(() {
        _isLoadingSurahs = false;
      });
    }
  }

  void _surahInfoFromJson(Map<String, dynamic> ch) {
    final int id = ch['id'] as int;
    final String nameArabic = ch['name_arabic'] as String;
    final int versesCount = ch['verses_count'] as int;

    _surahs.add(
      _SurahInfo(
        id: id,
        nameArabic: nameArabic,
        versesCount: versesCount,
      ),
    );
  }

  _SurahInfo? _findSurah(int? id) {
    if (id == null) return null;
    try {
      return _surahs.firstWhere((s) => s.id == id);
    } catch (_) {
      return null;
    }
  }

  int _maxAyahForSurah(int? surahId) {
    final info = _findSurah(surahId);
    return info?.versesCount ?? 1;
  }

    switch (value) {
      case 0:
        return 'بدون تكرار';
      case 1:
        return 'مرة واحدة';
      case 3:
        return '3 مرات';
      case 5:
        return '5 مرات';
      case 10:
        return '10 مرات';
      default:
        return '$value مرة';
    }
  }

  List<DropdownMenuItem<int>> _buildRepeatItems() {
    return _repeatOptions
        .map(
          (v) => DropdownMenuItem<int>(
            value: v,
            child: Text(_repeatLabelFor(v)),
          ),
        )
        .toList();
  }

        ),
        height: 460,
        child: Column(
          children: [

                  ? const Center(child: CircularProgressIndicator())
                  : Padding(
                      padding: const EdgeInsets.all(12.0),

                          final reciters = audioState.reciters;

                                            value: r,

                                      context

                                          .add(ChangeReciterEvent(reciter));
                                    },
                                    hint: 'اختر القارئ',
                                  ),
                                ),

                                // ------------- القسم: تحديد البدء -------------
                                MemorizationSectionCard(
                                  title: 'تحديد البدء',
                                  child: SurahAyahRow(
                                    surahDropdown: MemorizationDropdown<int>(
                                      value: _startSurahId,
                                      items: _surahs
                                          .map(
                                            (s) => DropdownMenuItem<int>(
                                              value: s.id,
                                              child:
                                                  Text('${s.id}. ${s.nameArabic}'),
                                            ),
                                          )
                                          .toList(),
                                      onChanged: (value) {
                                        setState(() {
                                          _startSurahId = value;
                                          _startAyah = 1;
                                        });
                                      },
                                      hint: 'اختر السورة',
                                    ),
                                    ayahDropdown: MemorizationDropdown<int>(
                                      value: _startAyah,
                                      items: List<int>.generate(
                                        _maxAyahForSurah(_startSurahId),
                                        (i) => i + 1,
                                      )
                                          .map(
                                            (v) => DropdownMenuItem<int>(
                                              value: v,
                                              child: Text('$v'),
                                            ),
                                          )
                                          .toList(),
                                      onChanged: (value) {
                                        setState(() {
                                          _startAyah = value;
                                        });
                                      },
                                      hint: 'آية',
                                    ),
                                  ),
                                ),

                                // ----------- القسم: تحديد الانتهاء ------------
                                MemorizationSectionCard(
                                  title: 'تحديد الانتهاء',
                                  child: SurahAyahRow(
                                    surahDropdown: MemorizationDropdown<int>(
                                      value: _endSurahId,
                                      items: _surahs
                                          .map(
                                            (s) => DropdownMenuItem<int>(
                                              value: s.id,
                                              child:
                                                  Text('${s.id}. ${s.nameArabic}'),
                                            ),
                                          )
                                          .toList(),
                                      onChanged: (value) {
                                        setState(() {
                                          _endSurahId = value;
                                          _endAyah =
                                              _maxAyahForSurah(_endSurahId);
                                        });
                                      },
                                      hint: 'اختر السورة',
                                    ),
                                    ayahDropdown: MemorizationDropdown<int>(
                                      value: _endAyah,
                                      items: List<int>.generate(
                                        _maxAyahForSurah(_endSurahId),
                                        (i) => i + 1,
                                      )
                                          .map(
                                            (v) => DropdownMenuItem<int>(
                                              value: v,
                                              child: Text('$v'),
                                            ),
                                          )
                                          .toList(),
                                      onChanged: (value) {
                                        setState(() {
                                          _endAyah = value;
                                        });
                                      },
                                      hint: 'آية',
                                    ),
                                  ),
                                ),

                                // ---------------- القسم: التكرار ----------------
                                MemorizationSectionCard(
                                  title: 'التكرار',
                                  child: Column(
                                    children: [
                                      RepeatOptionRow<int>(
                                        label: 'تكرار الآية',
                                        value: _ayahRepeatCount,
                                        items: _buildRepeatItems(),
                                        onChanged: (value) {
                                          setState(() {
                                            _ayahRepeatCount = value ?? 0;
                                          });
                                        },
                                      ),
                                      RepeatOptionRow<int>(
                                        label: 'تكرار الفقرة',
                                        value: _rangeRepeatCount,
                                        items: _buildRepeatItems(),
                                        onChanged: (value) {
                                          setState(() {
                                            _rangeRepeatCount = value ?? 0;
                                          });
                                        },

                                    ],

                            ),
                          );
                        },
                      ),
                    ),

            // ----------------- زر "بدء" في الأسفل -----------------
            Padding(
              padding:
                  const EdgeInsets.symmetric(horizontal: 12.0, vertical: 8),
              child: MemorizationStartButton(
                onPressed: _onStartPressed,
              ),
            ),
          ],
        ),
      ),
    );
  }

  void _onStartPressed() {
    final audioBloc = context.read<QuranAudioBloc>();

    if (_startSurahId == null ||
        _endSurahId == null ||
        _startAyah == null ||
        _endAyah == null) {
      Navigator.pop(context);
      return;
    }

    // حالياً: نقيّد التشغيل على نفس السورة فقط (لتبسيط المنطق)
    int surahNumber = _startSurahId!;
    if (_endSurahId != _startSurahId) {
      // لو اختلفت السور، نخلي النهاية ضمن نفس السورة
      _endSurahId = _startSurahId;
    }

    int startAyah = _startAyah!;
    int endAyah = _endAyah!;

    if (endAyah < startAyah) {
      final tmp = startAyah;
      startAyah = endAyah;
      endAyah = tmp;
    }

    // لو تكرار الفقرة = 0 نعتبره مرة واحدة على الأقل
    final int repeatCount = _rangeRepeatCount == 0 ? 1 : _rangeRepeatCount;

    audioBloc.add(
      PlayAudioRangeEvent(
        surahNumber: surahNumber,
        startAyah: startAyah,
        endAyah: endAyah,
        repeatCount: repeatCount,
      ),
    );

    Navigator.pop(context);
  }
}
